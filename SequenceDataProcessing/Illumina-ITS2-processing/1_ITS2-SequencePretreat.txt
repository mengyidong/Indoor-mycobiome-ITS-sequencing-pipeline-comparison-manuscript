###This file contains the pretreatment of the demultiplexed sequence data.
###Setup of working directory - change this path to match your file system

WD_path=/your/project/folder
mkdir ${WD_path}
cd ${WD_path}


###Set up biom-format if you have not
conda install biom-format


###Generates filelist to loop through, including only Mengyi's samples which were also on this run

ls /your/data/folder | grep "R1_001.fastq.gz" | grep "ProjectName" | sed 's/_R1_001.fastq.gz//g' > ${WD_path}/filelist


###Uses PEAR to merge paired reads

mkdir ${WD_path}/PEARReads/ && cd ${WD_path}/PEARReads/

for i in $(cat ${WD_path}/filelist)
do
sbatch -o slurm-%j-PEAR.out --partition=dmcshared --export=ALL -t 2:00:00 -c 1 --wrap="pear \
        -f /your/data/folder/${I}_R1_001.fastq.gz \
        -r /your/data/folder/${I}_R2_001.fastq.gz \
        -o ${WD_path}/PEARReads/${i}"
done


###Uses ITSxpress to extract ITS2 region from merged reads

conda create -n itsxpressenv -c bioconda -c conda-forge itsxpress
conda activate itsxpressenv 	

mkdir ${WD_path}/ITSxpressReads/ && cd ${WD_path}/ITSxpressReads/

for i in $(cat ${WD_path}/filelist)
do
sbatch -o slurm-%j-ITSxpress.out --partition=dmcshared --export=ALL -t 24:00:00 -c 64 --wrap="itsxpress --fastq ${WD_path}/PEARReads/${i}.assembled.fastq --single_end --region ITS2 --taxa Fungi \
--log ${WD_path}/ITSxpressReads/${i}.logfile.txt --outfile ${WD_path}/ITSxpressReads/${i}.ITS.fastq.gz --threads 64"
done


###Prepares QIIME2 manifest
printf "%s\t%s\n" "sample-id" "absolute-filepath" > ${WD_path}/QIIMEManifest.tsv
for i in $(cat ${WD_path}/filelist)
do
SampleID=$(echo "$i" | grep -oP 'PreMiEr.')
printf "%s\t%s\n" "${SampleID}" "${WD_path}/ITSxpressReads/${i}.ITS.fastq.gz" >> ${WD_path}/QIIMEManifest.tsv
done
